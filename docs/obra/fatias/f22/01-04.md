CREATE TABLE agenda_escala (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  agenda_servico_id BIGINT UNSIGNED NOT NULL,
  dia_semana        TINYINT UNSIGNED NOT NULL COMMENT '1 = domingo ... 7 = sábado',
  hora_inicio       TIME NOT NULL COMMENT 'Hora local da agenda',
  hora_fim          TIME NOT NULL,
  intervalo_inicio  TIME NULL COMMENT 'Pausa dentro da faixa (almoço), sem horários',
  intervalo_fim     TIME NULL,
  duracao_horario_minutos SMALLINT UNSIGNED NULL COMMENT 'Duração nesta faixa; nulo usa a da agenda',
  vagas_por_horario SMALLINT UNSIGNED NOT NULL DEFAULT 1,
  vagas_encaixe     SMALLINT UNSIGNED NOT NULL DEFAULT 0 COMMENT 'Encaixes permitidos na faixa',
  tipo_marcacao_id  BIGINT UNSIGNED NULL COMMENT 'Tipo reservado para a faixa; nulo aceita todos',
  recurso_agendavel_id BIGINT UNSIGNED NULL,
  local_atendimento_id BIGINT UNSIGNED NULL,
  vigencia_inicio   DATE NOT NULL,
  vigencia_fim      DATE NULL,
  atende_feriado    BOOLEAN NOT NULL DEFAULT FALSE,
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_agenda_escala_dia_semana CHECK (dia_semana BETWEEN 1 AND 7),
  CONSTRAINT ck_agenda_escala_horas CHECK (hora_fim > hora_inicio),
  CONSTRAINT ck_agenda_escala_intervalo CHECK (
    (intervalo_inicio IS NULL AND intervalo_fim IS NULL)
    OR (intervalo_inicio IS NOT NULL AND intervalo_fim IS NOT NULL AND intervalo_inicio >= hora_inicio
        AND intervalo_fim > intervalo_inicio AND intervalo_fim <= hora_fim)),
  CONSTRAINT ck_agenda_escala_duracao CHECK (duracao_horario_minutos IS NULL OR duracao_horario_minutos BETWEEN 5 AND 480),
  CONSTRAINT ck_agenda_escala_vagas CHECK (vagas_por_horario >= 1),
  CONSTRAINT ck_agenda_escala_vigencia CHECK (vigencia_fim IS NULL OR vigencia_fim >= vigencia_inicio),
  KEY ix_agenda_escala_agenda_dia (agenda_servico_id, dia_semana, hora_inicio),
  KEY ix_agenda_escala_tipo_marcacao (tipo_marcacao_id),
  KEY ix_agenda_escala_recurso (recurso_agendavel_id),
  KEY ix_agenda_escala_local (local_atendimento_id)
) COMMENT='Escala semanal da agenda: dia, hora inicial e final, intervalo e vagas por faixa';

  ADD CONSTRAINT fk_agenda_escala_recurso_agendavel_id FOREIGN KEY (recurso_agendavel_id) REFERENCES recurso_agendavel (id) ON DELETE RESTRICT,
