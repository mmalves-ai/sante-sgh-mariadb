  estabelecimento_id                    BIGINT UNSIGNED NOT NULL,
  local_estoque_id                      BIGINT UNSIGNED NOT NULL,
  lote_insumo_id                        BIGINT UNSIGNED NOT NULL COMMENT 'Lote (e por ele o insumo e a validade)',
  endereco_armazem_id                   BIGINT UNSIGNED NULL,
  quantidade                            DECIMAL(15,4) NOT NULL DEFAULT 0,
  quantidade_reservada                  DECIMAL(15,4) NOT NULL DEFAULT 0,
  quantidade_consignada                 DECIMAL(15,4) NOT NULL DEFAULT 0,
  quantidade_doada                      DECIMAL(15,4) NOT NULL DEFAULT 0,
  quantidade_em_kit                     DECIMAL(15,4) NOT NULL DEFAULT 0,
  competencia                           DATE NOT NULL COMMENT 'Mês corrente do saldo (primeiro dia)',
  quantidade_inicial_competencia        DECIMAL(15,4) NOT NULL DEFAULT 0,
  quantidade_final_competencia_anterior DECIMAL(15,4) NULL,
  bloqueado                             BOOLEAN NOT NULL DEFAULT FALSE,
  bloqueio_lote_id                      BIGINT UNSIGNED NULL,
  codigo_identificador_medicamento      VARCHAR(100) NULL COMMENT 'Identificador único do medicamento (rastreabilidade sanitária)',
  ultima_movimentacao_em                DATETIME(3) NULL,
  origem                                VARCHAR(20) NOT NULL DEFAULT 'sistema',
  criado_em                             DATETIME(3) NOT NULL,
  criado_por                            BIGINT UNSIGNED NULL,
  alterado_em                           DATETIME(3) NULL,
  alterado_por                          BIGINT UNSIGNED NULL,
  versao                                INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em                           DATETIME(3) NULL,
  excluido_por                          BIGINT UNSIGNED NULL,
  CONSTRAINT ck_saldo_lote_origem CHECK (origem IN ('sistema','migracao','integracao')),
  CONSTRAINT ck_saldo_lote_quantidades CHECK (quantidade >= 0 AND quantidade_reservada >= 0 AND quantidade_consignada >= 0 AND quantidade_doada >= 0 AND quantidade_em_kit >= 0),
  CONSTRAINT ck_saldo_lote_competencia CHECK (DAYOFMONTH(competencia) = 1),
  CONSTRAINT ck_saldo_lote_bloqueio CHECK (bloqueado = FALSE OR bloqueio_lote_id IS NOT NULL),
  UNIQUE KEY uq_saldo_lote_local_estoque_id_lote_insumo_id (local_estoque_id, lote_insumo_id),
  KEY ix_saldo_lote_estabelecimento_id (estabelecimento_id),
  KEY ix_saldo_lote_lote_insumo_id (lote_insumo_id),
  KEY ix_saldo_lote_endereco_armazem_id (endereco_armazem_id),
  KEY ix_saldo_lote_bloqueio_lote_id (bloqueio_lote_id)
) COMMENT='Saldo de cada lote por local, com saldo inicial e final do mês';
