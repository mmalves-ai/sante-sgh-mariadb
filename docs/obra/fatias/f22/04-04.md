CREATE TABLE relatorio_parametro_pesquisa (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,

  entidade          VARCHAR(64) NOT NULL COMMENT 'Fonte de dados da pesquisa, nomeada pelo produto (ex.: fonte_pagadora, unidade_assistencial)',

  campo_complemento VARCHAR(64) NULL COMMENT 'Campo extra mostrado na lista',
  filtro            JSON NULL COMMENT 'Filtro fixo da pesquisa (ex.: só ativos, só do estabelecimento da sessão)',
  retorno           VARCHAR(30) NOT NULL DEFAULT 'unico',
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_relatorio_parametro_pesquisa_retorno CHECK (retorno IN ('unico','multiplo')),
  CONSTRAINT ck_relatorio_parametro_pesquisa_filtro CHECK (filtro IS NULL OR JSON_VALID(filtro)),
  UNIQUE KEY uq_relatorio_parametro_pesquisa_codigo (codigo)
) COMMENT='Pesquisa que alimenta um parâmetro de relatório (lista de convênios, unidades), com filtro';
