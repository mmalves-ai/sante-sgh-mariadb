## 4. DDL de destino (de `db/migracoes`)

### geracao_grade (`db/migracoes/05_AGE/01_tabelas.sql`)

```sql
CREATE TABLE geracao_grade (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  estabelecimento_id BIGINT UNSIGNED NOT NULL,
  agenda_servico_id BIGINT UNSIGNED NOT NULL,
  periodo_inicio    DATE NOT NULL,
  periodo_fim       DATE NOT NULL,
  duracao_horario_minutos SMALLINT UNSIGNED NULL COMMENT 'Duração usada nesta geração; nulo usa a da escala ou da agenda',
  inclui_feriados   BOOLEAN NOT NULL DEFAULT FALSE,
  sobrepoe_existentes BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Substitui horários livres já gerados no período',
  liberacao_em      DATE NULL COMMENT 'Data a partir da qual os horários gerados ficam abertos para marcação',
  situacao          VARCHAR(30) NOT NULL DEFAULT 'pendente',
  solicitada_em     DATETIME(3) NOT NULL,
  responsavel_usuario_id BIGINT UNSIGNED NOT NULL,
  processada_em     DATETIME(3) NULL,
  horarios_gerados  INT UNSIGNED NOT NULL DEFAULT 0,
  revertida_em      DATETIME(3) NULL,
  reversao_usuario_id BIGINT UNSIGNED NULL,
  assinatura_parametros CHAR(32) NULL COMMENT 'Hash dos parâmetros, para conferir a reversão e evitar geração repetida',
  mensagem_processo VARCHAR(2000) NULL,
  origem            VARCHAR(20) NOT NULL DEFAULT 'sistema',
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em       DATETIME(3) NULL,
  excluido_por      BIGINT UNSIGNED NULL,
  CONSTRAINT ck_geracao_grade_periodo CHECK (periodo_fim >= periodo_inicio),
  CONSTRAINT ck_geracao_grade_duracao CHECK (duracao_horario_minutos IS NULL OR duracao_horario_minutos BETWEEN 5 AND 480),
```
