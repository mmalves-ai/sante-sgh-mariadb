  deposito_montante_integral   BOOLEAN NOT NULL DEFAULT FALSE,
  vigencia_inicio              DATE NOT NULL,
  vigencia_fim                 DATE NULL,
  situacao                     VARCHAR(30) NOT NULL DEFAULT 'ativo',
  origem                       VARCHAR(20) NOT NULL DEFAULT 'sistema',
  criado_em                    DATETIME(3) NOT NULL,
  criado_por                   BIGINT UNSIGNED NULL,
  alterado_em                  DATETIME(3) NULL,
  alterado_por                 BIGINT UNSIGNED NULL,

  excluido_em                  DATETIME(3) NULL,
  excluido_por                 BIGINT UNSIGNED NULL,
  UNIQUE KEY uq_processo_tributario_numero (estabelecimento_id, tipo, numero_processo),
  CONSTRAINT ck_processo_tributario_tipo CHECK (tipo IN ('administrativo','judicial','numero_beneficio','fap')),
  CONSTRAINT ck_processo_tributario_autoria CHECK (autoria IN ('proprio_contribuinte','outra_entidade')),
  CONSTRAINT ck_processo_tributario_situacao CHECK (situacao IN ('ativo','suspenso','encerrado')),
  CONSTRAINT ck_processo_tributario_vara CHECK (tipo <> 'judicial' OR codigo_vara IS NOT NULL),

) COMMENT='Processo administrativo ou judicial que suspende tributo, com número, vara e decisão';
