CREATE TABLE dia_nao_util (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  descricao         VARCHAR(150) NOT NULL,
  abrangencia       VARCHAR(30) NOT NULL,
  unidade_federativa_id BIGINT UNSIGNED NULL COMMENT 'Obrigatória na abrangência estadual',
  municipio_ibge_id BIGINT UNSIGNED NULL COMMENT 'Obrigatório na abrangência municipal',
  estabelecimento_id BIGINT UNSIGNED NULL COMMENT 'Obrigatório na abrangência de estabelecimento',
  recorrencia       VARCHAR(20) NOT NULL COMMENT 'anual: repete todo ano em dia e mês; data_unica: só na data; movel: data calculada a cada ano e gravada por ano',
  dia               TINYINT UNSIGNED NULL COMMENT 'Dia do mês, na recorrência anual',
  mes               TINYINT UNSIGNED NULL COMMENT 'Mês, na recorrência anual',
  data              DATE NULL COMMENT 'Data, na recorrência de data única ou móvel',
  facultativo       BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Ponto facultativo',
  hora_inicio       TIME NULL COMMENT 'Início do período sem expediente, quando não é o dia inteiro',
  hora_fim          TIME NULL,
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  origem            VARCHAR(20) NOT NULL DEFAULT 'sistema',
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em       DATETIME(3) NULL,
  excluido_por      BIGINT UNSIGNED NULL,
  CONSTRAINT ck_dia_nao_util_abrangencia CHECK (abrangencia IN ('nacional','estadual','municipal','estabelecimento')),
  CONSTRAINT ck_dia_nao_util_escopo CHECK (
    (abrangencia = 'nacional' AND unidade_federativa_id IS NULL AND municipio_ibge_id IS NULL AND estabelecimento_id IS NULL)
    OR (abrangencia = 'estadual' AND unidade_federativa_id IS NOT NULL AND municipio_ibge_id IS NULL AND estabelecimento_id IS NULL)
    OR (abrangencia = 'municipal' AND municipio_ibge_id IS NOT NULL AND estabelecimento_id IS NULL)
    OR (abrangencia = 'estabelecimento' AND estabelecimento_id IS NOT NULL)),
  CONSTRAINT ck_dia_nao_util_recorrencia CHECK (
    (recorrencia = 'anual' AND dia BETWEEN 1 AND 31 AND mes BETWEEN 1 AND 12 AND data IS NULL)
    OR (recorrencia IN ('data_unica','movel') AND data IS NOT NULL AND dia IS NULL AND mes IS NULL)),
  CONSTRAINT ck_dia_nao_util_horario CHECK ((hora_inicio IS NULL AND hora_fim IS NULL) OR (hora_inicio IS NOT NULL AND hora_fim IS NOT NULL AND hora_fim > hora_inicio)),
  CONSTRAINT ck_dia_nao_util_origem CHECK (origem IN ('sistema','migracao','integracao')),
  KEY ix_dia_nao_util_data (data),
  KEY ix_dia_nao_util_dia_mes (mes, dia),
  KEY ix_dia_nao_util_unidade_federativa (unidade_federativa_id),
  KEY ix_dia_nao_util_municipio (municipio_ibge_id),
  KEY ix_dia_nao_util_estabelecimento (estabelecimento_id)
) COMMENT='Feriado ou dia sem expediente com abrangência (nacional, estadual, municipal, estabelecimento), fixo ou móvel';
