```sql
CREATE TABLE tipo_atendimento_tabela (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  codigo          VARCHAR(30) NOT NULL,
  descricao       VARCHAR(150) NOT NULL,
  ativo           BOOLEAN NOT NULL DEFAULT TRUE,
  padrao          VARCHAR(20) NOT NULL COMMENT 'Padrão a que o tipo pertence',
  classe          VARCHAR(20) NULL,
  codigo_externo  VARCHAR(10) NULL COMMENT 'Código do tipo no padrão de origem (SUS ou TUSS)',
  criado_em       DATETIME(3) NOT NULL,
  criado_por      BIGINT UNSIGNED NULL,
  alterado_em     DATETIME(3) NULL,
  alterado_por    BIGINT UNSIGNED NULL,
  versao          INT UNSIGNED NOT NULL DEFAULT 1,
  UNIQUE KEY uq_tipo_atendimento_tabela_codigo (codigo),
  CONSTRAINT ck_tipo_atendimento_tabela_padrao CHECK (padrao IN ('sus','tuss','presencial','proprio')),
  CONSTRAINT ck_tipo_atendimento_tabela_classe CHECK (classe IS NULL OR classe IN ('consulta','exame','internacao','procedimento','terapia','outro')),
  UNIQUE KEY uq_tipo_atendimento_tabela_padrao_externo (padrao, codigo_externo)
) COMMENT='Tipos de atendimento a que um item da tabela se aplica (consulta, exame, internação, presencial)';
```
