```sql
CREATE TABLE tipo_certificado (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  codigo            VARCHAR(30) NOT NULL,
  descricao         VARCHAR(150) NOT NULL,
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  modalidade        VARCHAR(30) NOT NULL,
  repositorio       VARCHAR(30) NOT NULL COMMENT 'Onde a chave privada fica',
  icp_brasil        BOOLEAN NOT NULL DEFAULT TRUE COMMENT 'Assinatura qualificada da ICP-Brasil',
  exige_provedor    BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Precisa de provedor em nuvem configurado',
  validade_maxima_meses SMALLINT UNSIGNED NULL,
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  UNIQUE KEY uq_tipo_certificado_codigo (codigo),
  CONSTRAINT ck_tipo_certificado_modalidade CHECK (modalidade IN ('a1','a3','nuvem','avancada','simples')),
  CONSTRAINT ck_tipo_certificado_repositorio CHECK (repositorio IN ('arquivo','cartao','token','hsm','nuvem','nenhum'))
) COMMENT='Tipos de certificado aceitos (A1, A3, em nuvem, assinatura eletrônica avançada)';
```

```sql
CREATE TABLE provedor_assinatura_nuvem (

  ambiente          VARCHAR(30) NOT NULL DEFAULT 'producao',
  homologado        BOOLEAN NOT NULL DEFAULT FALSE,
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_provedor_assinatura_nuvem_ambiente CHECK (ambiente IN ('producao','homologacao')),
  UNIQUE KEY uq_provedor_assinatura_nuvem_codigo (codigo),
  KEY ix_provedor_assinatura_nuvem_estabelecimento (estabelecimento_id),
  KEY ix_provedor_assinatura_nuvem_tipo_certificado (tipo_certificado_id)
) COMMENT='Provedor de certificado em nuvem homologado, com endereço do serviço e referência da credencial no cofre';
```

```sql
CREATE TABLE oid_icp (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  codigo            VARCHAR(30) NOT NULL,
  descricao         VARCHAR(150) NOT NULL,
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,

  conselho_classe_id BIGINT UNSIGNED NULL COMMENT 'Conselho profissional, quando o identificador traz registro em conselho',
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_oid_icp_tipo CHECK (tipo_oid IN ('politica','cpf','cnpj','nome','data_nascimento','titulo_eleitor','registro_conselho','uf_conselho','especialidade','cnes','responsavel','outro')),
  UNIQUE KEY uq_oid_icp_codigo (codigo),
  UNIQUE KEY uq_oid_icp_oid (oid),
  KEY ix_oid_icp_conselho (conselho_classe_id)
) COMMENT='Identificadores de política e de atributos do certificado da infraestrutura nacional, usados na validação';
```
