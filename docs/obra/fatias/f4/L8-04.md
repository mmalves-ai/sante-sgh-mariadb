```sql
CREATE TABLE preco_procedimento (
  id                     BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  tabela_preco_id        BIGINT UNSIGNED NOT NULL,
  sequencia              INT UNSIGNED NOT NULL,
  item_faturavel_id      BIGINT UNSIGNED NOT NULL,
  codigo_cobranca        VARCHAR(20) NULL COMMENT 'Código do procedimento nesta tabela, quando difere do item',
  descricao_cobranca     VARCHAR(150) NULL,
  modelo_valor           VARCHAR(20) NOT NULL DEFAULT 'valor' COMMENT 'Como o valor é formado: valor direto, porte ou índice',
  porte_procedimento_id  BIGINT UNSIGNED NULL COMMENT 'Porte de honorário',
  porte_anestesico_id    BIGINT UNSIGNED NULL,
  quantidade_indice      DECIMAL(15,4) NULL COMMENT 'Quantidade de índice de cobrança (CH)',
  quantidade_filme_m2    DECIMAL(15,4) NULL COMMENT 'Metragem de filme em m²',
  incidencias            SMALLINT UNSIGNED NULL COMMENT 'Incidências de imagem',
  quantidade_auxiliares  TINYINT UNSIGNED NOT NULL DEFAULT 0,
  quantidade_maxima      DECIMAL(15,4) NULL COMMENT 'Quantidade máxima por lançamento',
  termo_versao_tiss_id   BIGINT UNSIGNED NULL COMMENT 'Equivalência na terminologia TUSS',
  ativo                  BOOLEAN NOT NULL DEFAULT TRUE,
  origem                 VARCHAR(20) NOT NULL DEFAULT 'sistema',
  criado_em              DATETIME(3) NOT NULL,
  criado_por             BIGINT UNSIGNED NULL,
  alterado_em            DATETIME(3) NULL,
  alterado_por           BIGINT UNSIGNED NULL,
  versao                 INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em            DATETIME(3) NULL,
  excluido_por           BIGINT UNSIGNED NULL,
  CONSTRAINT ck_preco_procedimento_origem CHECK (origem IN ('sistema','migracao','integracao')),
  CONSTRAINT ck_preco_procedimento_modelo CHECK (modelo_valor IN ('valor','porte','indice')),
  CONSTRAINT ck_preco_procedimento_porte CHECK (modelo_valor <> 'porte' OR porte_procedimento_id IS NOT NULL),
  CONSTRAINT ck_preco_procedimento_indice CHECK (modelo_valor <> 'indice' OR quantidade_indice IS NOT NULL),
  UNIQUE KEY uq_preco_procedimento_tabela_sequencia (tabela_preco_id, sequencia),
  UNIQUE KEY uq_preco_procedimento_tabela_item (tabela_preco_id, item_faturavel_id),
  KEY ix_preco_procedimento_item_faturavel_id (item_faturavel_id),
  KEY ix_preco_procedimento_porte_procedimento_id (porte_procedimento_id),
  KEY ix_preco_procedimento_porte_anestesico_id (porte_anestesico_id),
  KEY ix_preco_procedimento_termo_versao_tiss_id (termo_versao_tiss_id)
) COMMENT='Procedimento dentro da tabela: código, porte, quantidade de índices, filme, auxiliares e equivalência TUSS';
```
