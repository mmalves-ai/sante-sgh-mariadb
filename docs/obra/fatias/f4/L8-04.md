CREATE TABLE equivalencia_matmed_tuss (

  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em       DATETIME(3) NULL,
  excluido_por      BIGINT UNSIGNED NULL,
  CONSTRAINT ck_equivalencia_matmed_tuss_vigencia CHECK (vigencia_fim IS NULL OR vigencia_fim >= vigencia_inicio),
  CONSTRAINT ck_equivalencia_matmed_tuss_tipo CHECK (tipo_equivalencia IS NULL OR tipo_equivalencia IN ('exata','aproximada','generica')),
  UNIQUE KEY uq_equivalencia_matmed_tuss_par (item_referencia_matmed_id, termo_terminologia_id, vigencia_inicio),
