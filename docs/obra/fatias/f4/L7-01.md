### hemocomponente (`db/migracoes/10_SADT/02_tabelas.sql`)

```sql
CREATE TABLE hemocomponente (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  codigo            VARCHAR(30) NOT NULL,
  descricao         VARCHAR(150) NOT NULL,
  sigla             VARCHAR(4) NOT NULL COMMENT 'Sigla usada na etiqueta e na entrada da bolsa (ex.: CH, CP, PFC, CRIO)',
  natureza          VARCHAR(30) NOT NULL DEFAULT 'hemocomponente',
  unidade_medida_id BIGINT UNSIGNED NULL,
  volume_padrao_ml  SMALLINT UNSIGNED NULL,
  validade_dias     SMALLINT UNSIGNED NOT NULL COMMENT 'Validade a partir da coleta ou do preparo',
  conservacao       VARCHAR(30) NOT NULL DEFAULT 'refrigerado',
  temperatura_minima DECIMAL(5,1) NULL COMMENT 'Temperatura mínima de guarda, em graus Celsius',
  temperatura_maxima DECIMAL(5,1) NULL COMMENT 'Temperatura máxima de guarda, em graus Celsius',
  exige_prova_compatibilidade BOOLEAN NOT NULL DEFAULT FALSE,
  exige_agitacao    BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Guarda sob agitação contínua (plaquetas)',
  faturavel         BOOLEAN NOT NULL DEFAULT TRUE,
  item_faturavel_id BIGINT UNSIGNED NULL COMMENT 'Item cobrado na transfusão',
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  UNIQUE KEY uq_hemocomponente_codigo (codigo),
  CONSTRAINT ck_hemocomponente_natureza CHECK (natureza IN ('hemocomponente','hemoderivado')),
  CONSTRAINT ck_hemocomponente_conservacao CHECK (conservacao IN ('ambiente','refrigerado','congelado')),
  CONSTRAINT ck_hemocomponente_temperatura CHECK (temperatura_minima IS NULL OR temperatura_maxima IS NULL OR temperatura_minima <= temperatura_maxima),
  CONSTRAINT ck_hemocomponente_validade CHECK (validade_dias BETWEEN 1 AND 3650),
  CONSTRAINT ck_hemocomponente_faturavel CHECK (faturavel = FALSE OR item_faturavel_id IS NOT NULL),
  UNIQUE KEY uq_hemocomponente_sigla (sigla),
  KEY ix_hemocomponente_unidade_medida (unidade_medida_id),
  KEY ix_hemocomponente_item_faturavel (item_faturavel_id)
) COMMENT='Tipo de hemocomponente ou hemoderivado (concentrado de hemácias, plaquetas, plasma, crioprecipitado) com validade e conservação';
```
