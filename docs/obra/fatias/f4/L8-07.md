  criado_em DATETIME(3) NOT NULL,
  criado_por BIGINT UNSIGNED NULL,
  alterado_em DATETIME(3) NULL,
  alterado_por BIGINT UNSIGNED NULL,
  versao INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_dominio_sus_lista CHECK (lista IN ('motivo_saida','carater_atendimento','credor','classificacao_diagnostico','justificativa_paciente_sem_documento','tipo_apac','local_atendimento','associacao_quimioterapia','regional_saude','tipo_leito_uti','acesso_vascular','caracteristica_tratamento_dialitico','contraindicacao_transplante','situacao_transplante','grupo_atendimento','tipo_atendimento','tipo_prestador','nivel_atencao','fonte_orcamentaria','contraceptivo','finalidade_tratamento','grupo_justificativa_auditiva','justificativa_auditiva','medicamento_antineoplasico','local_metastase','atividade_profissional','estadio_tumor')),
