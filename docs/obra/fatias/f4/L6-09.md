```sql
CREATE TABLE gravidade_interacao (

  criado_em                    DATETIME(3) NOT NULL,
  criado_por                   BIGINT UNSIGNED NULL,
  alterado_em                  DATETIME(3) NULL,
  alterado_por                 BIGINT UNSIGNED NULL,
  versao                       INT UNSIGNED NOT NULL DEFAULT 1,
  UNIQUE KEY uq_gravidade_interacao_codigo (codigo),
  CONSTRAINT ck_gravidade_interacao_cor CHECK (cor IS NULL OR cor REGEXP '^#[0-9A-Fa-f]{6}$'),
  UNIQUE KEY uq_gravidade_interacao_nivel (nivel),
  KEY ix_gravidade_interacao_acao_gravidade_interacao_id (acao_gravidade_interacao_id)
```

```sql
CREATE TABLE campo_autopreenchido_prescricao (
  id                       BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  estabelecimento_id       BIGINT UNSIGNED NULL COMMENT 'Nulo: vale para todos os estabelecimentos do cliente',
  setor_funcional_id       BIGINT UNSIGNED NULL COMMENT 'Nulo: vale para todos os setores',
  campo_destino            VARCHAR(40) NOT NULL COMMENT 'Campo da prescrição preenchido',
  fonte_dado               VARCHAR(40) NOT NULL COMMENT 'Dado de onde vem o valor',
  sobrescreve_digitado     BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Substitui o valor já digitado pelo prescritor',
  permite_editar           BOOLEAN NOT NULL DEFAULT TRUE COMMENT 'O prescritor pode alterar o valor preenchido',
  validade_horas           SMALLINT UNSIGNED NULL COMMENT 'Idade máxima do dado de origem (ex.: peso aferido há até 24 horas)',
  ativo                    BOOLEAN NOT NULL DEFAULT TRUE,
  criado_em           DATETIME(3) NOT NULL,
  criado_por          BIGINT UNSIGNED NULL,
  alterado_em         DATETIME(3) NULL,
  alterado_por        BIGINT UNSIGNED NULL,
  versao              INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_campo_autopreenchido_prescricao_destino CHECK (campo_destino IN ('peso','altura','superficie_corporal','alergias','diagnostico_principal','clearance_creatinina','idade','dieta','observacao')),
  CONSTRAINT ck_campo_autopreenchido_prescricao_fonte CHECK (fonte_dado IN ('ultima_afericao','cadastro_paciente','alergia_paciente','diagnostico_atendimento','resultado_exame','prescricao_anterior','calculo')),
  UNIQUE KEY uq_campo_autopreenchido_prescricao_campo (estabelecimento_id, setor_funcional_id, campo_destino),
  KEY ix_campo_autopreenchido_prescricao_setor_funcional (setor_funcional_id)
) COMMENT='Campos da prescrição preenchidos automaticamente a partir de outro dado (peso, alergia, diagnóstico)';
```

### tipo_encaminhamento (`db/migracoes/06_PEP/11_tabelas.sql`)

```sql
CREATE TABLE tipo_encaminhamento (
  id                   BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  codigo               VARCHAR(30) NOT NULL,
  descricao            VARCHAR(150) NOT NULL,
  ativo                BOOLEAN NOT NULL DEFAULT TRUE,
  direcao              VARCHAR(30) NOT NULL,
  exige_destino        BOOLEAN NOT NULL DEFAULT TRUE COMMENT 'Exige informar o serviço ou profissional de destino',
  criado_em            DATETIME(3) NOT NULL,
  criado_por           BIGINT UNSIGNED NULL,
  alterado_em          DATETIME(3) NULL,
  alterado_por         BIGINT UNSIGNED NULL,
  versao               INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_tipo_encaminhamento_direcao CHECK (direcao IN ('interno','externo','contrarreferencia','pos_alta')),
  UNIQUE KEY uq_tipo_encaminhamento_codigo (codigo)
) COMMENT='Tipos de encaminhamento (interno, externo, contrarreferência)';
```
