CREATE TABLE aliquota_rat_estabelecimento (
  id                   BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  estabelecimento_id   BIGINT UNSIGNED NOT NULL,
  vigencia_inicio      DATE NOT NULL,
  vigencia_fim         DATE NULL,

  aliquota_rat         DECIMAL(7,4) NOT NULL,
  fap                  DECIMAL(7,4) NOT NULL COMMENT 'Fator acidentário de prevenção (0,5 a 2,0)',
  rat_ajustado         DECIMAL(9,6) AS (aliquota_rat * fap) STORED COMMENT 'RAT multiplicado pelo FAP',
  rat_processo_tributario_id BIGINT UNSIGNED NULL,
  fap_processo_tributario_id BIGINT UNSIGNED NULL,
  criado_em            DATETIME(3) NOT NULL,
  criado_por           BIGINT UNSIGNED NULL,
  alterado_em          DATETIME(3) NULL,
  alterado_por         BIGINT UNSIGNED NULL,
  versao               INT UNSIGNED NOT NULL DEFAULT 1,
  UNIQUE KEY uq_aliquota_rat_estabelecimento_estabelecimento_vigencia_inicio (estabelecimento_id, vigencia_inicio),
  CONSTRAINT ck_aliquota_rat_estabelecimento_vigencia CHECK (vigencia_fim IS NULL OR vigencia_inicio IS NULL OR vigencia_fim >= vigencia_inicio),
  CONSTRAINT ck_aliquota_rat_estabelecimento_aliquota_rat CHECK (aliquota_rat IN (1, 2, 3)),
  CONSTRAINT ck_aliquota_rat_estabelecimento_fap CHECK (fap BETWEEN 0.5 AND 2.0),
  KEY ix_aliquota_rat_estabelecimento_rat_processo_tributario (rat_processo_tributario_id),
  KEY ix_aliquota_rat_estabelecimento_fap_processo_tributario (fap_processo_tributario_id)
) COMMENT='RAT e FAP do estabelecimento por vigência, com processo judicial se houver';
