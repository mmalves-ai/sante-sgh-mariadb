```sql
CREATE TABLE evento_repasse_contabil (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  evento_repasse_id BIGINT UNSIGNED NOT NULL,
  natureza_financeira_id BIGINT UNSIGNED NULL COMMENT 'Grupo e item financeiro',
  conta_contabil_id BIGINT UNSIGNED NULL,
  historico_padrao_id BIGINT UNSIGNED NULL,
  centro_custo_id   BIGINT UNSIGNED NULL,
  vigencia_inicio   DATE NOT NULL,
  vigencia_fim      DATE NULL COMMENT 'Nulo enquanto vigente',
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_evento_repasse_contabil_vigencia CHECK (vigencia_fim IS NULL OR vigencia_fim >= vigencia_inicio),
  CONSTRAINT ck_evento_repasse_contabil_destino CHECK (natureza_financeira_id IS NOT NULL OR conta_contabil_id IS NOT NULL),
  UNIQUE KEY uq_evento_repasse_contabil_evento (evento_repasse_id, vigencia_inicio),
```
