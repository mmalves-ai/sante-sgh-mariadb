## 4. DDL de destino (de `db/migracoes`)

### tributo (`db/migracoes/14_FIN/10_tabelas.sql`)

```sql
CREATE TABLE tributo (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  codigo            VARCHAR(30) NOT NULL COMMENT 'Sigla: IRRF, INSS, ISS, PIS, COFINS, CSLL, ICMS, IPI, IBS, CBS',
  descricao         VARCHAR(150) NOT NULL,
  ativo             BOOLEAN NOT NULL DEFAULT TRUE,
  esfera            VARCHAR(30) NOT NULL,
  forma_apuracao    VARCHAR(30) NOT NULL COMMENT 'Quando o tributo é apurado e recolhido',
  incidencia        VARCHAR(30) NOT NULL COMMENT 'Retido do terceiro, próprio do estabelecimento, ou ambos',
  codigo_receita    VARCHAR(10) NULL COMMENT 'Código de receita da guia de recolhimento (DARF, GPS, DAM)',
  valor_minimo_retencao DECIMAL(15,2) NULL COMMENT 'Abaixo dele a retenção não é feita (ex.: R$ 10,00 no IRRF)',
  acumula_abaixo_minimo BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Valor abaixo do mínimo acumula para a próxima retenção',
  dia_vencimento    TINYINT UNSIGNED NULL COMMENT 'Dia de vencimento do recolhimento no mês seguinte',
  natureza_financeira_id BIGINT UNSIGNED NULL COMMENT 'Natureza usada no título de recolhimento',
  conta_contabil_id BIGINT UNSIGNED NULL COMMENT 'Conta do passivo do tributo a recolher',
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  CONSTRAINT ck_tributo_esfera CHECK (esfera IN ('federal','estadual','municipal')),
  CONSTRAINT ck_tributo_forma_apuracao CHECK (forma_apuracao IN ('por_pagamento','por_operacao','mensal','trimestral','anual')),
  CONSTRAINT ck_tributo_incidencia CHECK (incidencia IN ('retido','proprio','ambos')),
  CONSTRAINT ck_tributo_minimo CHECK (valor_minimo_retencao IS NULL OR valor_minimo_retencao >= 0),
  CONSTRAINT ck_tributo_dia_vencimento CHECK (dia_vencimento IS NULL OR dia_vencimento BETWEEN 1 AND 31),
  UNIQUE KEY uq_tributo_codigo (codigo),
  KEY ix_tributo_natureza_financeira (natureza_financeira_id),
  KEY ix_tributo_conta_contabil (conta_contabil_id)
) COMMENT='Tributos tratados (IRRF, INSS, ISS, PIS, COFINS, CSLL, IBS, CBS), com esfera e forma de apuração';
```
