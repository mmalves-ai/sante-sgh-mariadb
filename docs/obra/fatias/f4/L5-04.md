CREATE TABLE endereco_pessoa (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  pessoa_fisica_id  BIGINT UNSIGNED NULL COMMENT 'Dono, quando pessoa física (exatamente um dos dois)',
  pessoa_juridica_id BIGINT UNSIGNED NULL COMMENT 'Dono, quando pessoa jurídica (exatamente um dos dois)',
  sequencia         SMALLINT UNSIGNED NOT NULL,
  tipo_endereco_id  BIGINT UNSIGNED NOT NULL COMMENT 'Finalidade: residencial, cobrança, correspondência, trabalho',
  exterior          BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Endereço fora do país',
  cep               CHAR(8) NULL,
  logradouro_cep_id BIGINT UNSIGNED NULL COMMENT 'Logradouro da base postal que completou o endereço',
  especie_logradouro_id BIGINT UNSIGNED NULL,
  logradouro        VARCHAR(150) NOT NULL,
  numero            VARCHAR(10) NULL,
  sem_numero        BOOLEAN NOT NULL DEFAULT FALSE,
  complemento       VARCHAR(100) NULL,
  bairro_id         BIGINT UNSIGNED NULL,
  bairro_nome       VARCHAR(100) NULL COMMENT 'Bairro digitado quando não está na base postal',
  municipio_ibge_id BIGINT UNSIGNED NULL COMMENT 'Obrigatório no endereço nacional',
  distrito_sanitario_id BIGINT UNSIGNED NULL,
  pais_referencia_id BIGINT UNSIGNED NULL COMMENT 'Obrigatório no endereço no exterior',
  cidade_exterior   VARCHAR(100) NULL,
  codigo_postal_exterior VARCHAR(20) NULL,
  ponto_referencia  VARCHAR(150) NULL,
  latitude          DECIMAL(9,6) NULL,
  longitude         DECIMAL(9,6) NULL,
  principal         BOOLEAN NOT NULL DEFAULT FALSE,
  vigencia_inicio   DATE NULL,
  vigencia_fim      DATE NULL,
  origem            VARCHAR(20) NOT NULL DEFAULT 'sistema',
  criado_em         DATETIME(3) NOT NULL,
  criado_por        BIGINT UNSIGNED NULL,
  alterado_em       DATETIME(3) NULL,
  alterado_por      BIGINT UNSIGNED NULL,
  versao            INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em       DATETIME(3) NULL,
  excluido_por      BIGINT UNSIGNED NULL,
  CONSTRAINT ck_endereco_pessoa_dono CHECK ((pessoa_fisica_id IS NULL) <> (pessoa_juridica_id IS NULL)),
  CONSTRAINT ck_endereco_pessoa_local CHECK ((NOT exterior AND cep IS NOT NULL AND municipio_ibge_id IS NOT NULL) OR (exterior AND pais_referencia_id IS NOT NULL AND cidade_exterior IS NOT NULL)),
  CONSTRAINT ck_endereco_pessoa_cep CHECK (cep IS NULL OR cep REGEXP '^[0-9]{8}$'),
  CONSTRAINT ck_endereco_pessoa_numero CHECK ((sem_numero AND numero IS NULL) OR (NOT sem_numero AND numero IS NOT NULL)),
  CONSTRAINT ck_endereco_pessoa_coordenada CHECK ((latitude IS NULL AND longitude IS NULL) OR (latitude BETWEEN -90 AND 90 AND longitude BETWEEN -180 AND 180)),
  CONSTRAINT ck_endereco_pessoa_vigencia CHECK (vigencia_fim IS NULL OR vigencia_inicio IS NULL OR vigencia_fim >= vigencia_inicio),
