```sql
CREATE TABLE resposta_padrao (
  id                          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  grupo_resposta_id           BIGINT UNSIGNED NOT NULL,
  sequencia                   SMALLINT UNSIGNED NOT NULL COMMENT 'Ordem de exibição no grupo',

  mnemonico                   VARCHAR(10) NULL COMMENT 'Atalho de digitação',
  anormal                     BOOLEAN NOT NULL DEFAULT FALSE,
  critico                     BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Dispara aviso de valor crítico',
  contagioso                  BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Indica doença transmissível (aciona controle de infecção)',
  exige_confirmacao           BOOLEAN NOT NULL DEFAULT FALSE,
  padrao                      BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Resposta sugerida na digitação',
  ativo                       BOOLEAN NOT NULL DEFAULT TRUE,
  origem                      VARCHAR(20) NOT NULL DEFAULT 'sistema' COMMENT 'sistema, migracao ou integracao',
  criado_em                   DATETIME(3) NOT NULL,
  criado_por                  BIGINT UNSIGNED NULL,
  alterado_em                 DATETIME(3) NULL,
  alterado_por                BIGINT UNSIGNED NULL,
  versao                      INT UNSIGNED NOT NULL DEFAULT 1,
  excluido_em                 DATETIME(3) NULL,
  excluido_por                BIGINT UNSIGNED NULL,
  CONSTRAINT ck_resposta_padrao_origem CHECK (origem IN ('sistema','migracao','integracao')),
  UNIQUE KEY uq_resposta_padrao_sequencia (grupo_resposta_id, sequencia),
  UNIQUE KEY uq_resposta_padrao_codigo (grupo_resposta_id, codigo)
) COMMENT='Resposta possível de um grupo de respostas, com ordem e indicação de anormalidade';
```

```sql
CREATE TABLE recipiente_coleta (

  tipo                       VARCHAR(20) NOT NULL DEFAULT 'tubo',
  cor_tampa                  VARCHAR(30) NULL,

  ordem_coleta               SMALLINT UNSIGNED NULL COMMENT 'Posição na ordem de coleta recomendada',
  detalhes                   VARCHAR(1000) NULL,
  modelo_etiqueta_id         BIGINT UNSIGNED NULL,
  exclusivo_subamostra       BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Usado só para alíquota, nunca na coleta primária',
  teste_complementar         BOOLEAN NOT NULL DEFAULT FALSE,
  uso_cultura_positiva       BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Recipiente para repique de cultura positiva',
  ativo                      BOOLEAN NOT NULL DEFAULT TRUE,
  criado_em                  DATETIME(3) NOT NULL,
  criado_por                 BIGINT UNSIGNED NULL,
  alterado_em                DATETIME(3) NULL,
  alterado_por               BIGINT UNSIGNED NULL,
  versao                     INT UNSIGNED NOT NULL DEFAULT 1,
  UNIQUE KEY uq_recipiente_coleta_codigo (codigo),
  CONSTRAINT ck_recipiente_coleta_tipo CHECK (tipo IN ('tubo','frasco','coletor','swab','lamina','seringa','bolsa','outro')),
  CONSTRAINT ck_recipiente_coleta_volume CHECK (volume_maximo_ml IS NULL OR volume_maximo_ml > 0),
  KEY ix_recipiente_coleta_modelo_etiqueta (modelo_etiqueta_id)
```
